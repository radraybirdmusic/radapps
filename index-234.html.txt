let aCtx, anal, virtualBuffer = null, activeSources = [];

// THE UNIVERSAL FILE INTAKE
document.getElementById('f-in').onchange = (e) => {
    const file = e.target.files[0];
    const reader = new FileReader();
    
    document.getElementById('diag').innerText = `INJECTING: ${file.name.toUpperCase()}`;
    
    reader.onload = async (event) => {
        if (!aCtx) aCtx = new (window.AudioContext || window.webkitAudioContext)();
        const rawData = event.target.result;
        
        // FORCE DECODE: It tries to find audio in anything
        try {
            aCtx.decodeAudioData(rawData, (buffer) => {
                virtualBuffer = buffer;
                document.getElementById('diag').innerText = "LOADED AS VALID WAVE";
            }, (err) => {
                document.getElementById('diag').innerText = "FORCE-GLITCH: RAW DATA MODE";
                // If it fails, we'll eventually add the raw-to-PCM bridge here
            });
        } catch (err) {
            document.getElementById('diag').innerText = "REJECTED BY BIOS";
        }
    };
    reader.readAsArrayBuffer(file);
};

function playRaw() {
    if (!virtualBuffer || !aCtx) return;
    const s = aCtx.createBufferSource();
    s.buffer = virtualBuffer;
    s.connect(aCtx.destination);
    s.start(0);
    activeSources.push(s);
}

function stopAll() { activeSources.forEach(s => s.stop()); activeSources = []; }
