<script>
let aCtx, anal, src, rec, chks = [], virtualCache = null, isL = false, isF = false, ints = {l:0.5, r:0.5}, isR = false, activeSources = [];
let currentArpPitch = 1.0;
const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d'), diag = document.getElementById('diag'), prog = document.getElementById('rec-progress');

document.getElementById('b-mic').onclick = async () => {
    document.getElementById('gate').style.display = 'none'; 
    document.getElementById('ui').style.display = 'block';
    try {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        anal = aCtx.createAnalyser(); anal.fftSize = 1024; 
        anal.connect(aCtx.destination);
        draw(); 
    } catch (err) { diag.innerText = "CONTEXT ERROR"; }
};

async function pulse(basePitch) {
    if (!virtualCache || !aCtx) return;
    if (aCtx.state === 'suspended') await aCtx.resume();
    
    const res = await fetch(virtualCache);
    const ab = await res.arrayBuffer();
    aCtx.decodeAudioData(ab, (buffer) => {
        const s = aCtx.createBufferSource(); 
        s.buffer = buffer; 
        s.playbackRate.value = basePitch * currentArpPitch;
        s.connect(anal);
        s.start(0);
        activeSources.push(s); 
    });
}

async function hamMomentum(multiplier) {
    currentArpPitch *= multiplier;
    diag.innerText = `OCTAVE: ${currentArpPitch.toFixed(2)}`;
    // If a file is currently playing, this will affect the NEXT trigger
}

document.getElementById('b-rec').onclick = () => {
    if (rec && rec.state === "recording") return;
    // Simple Recorder Logic
    diag.innerText = "RECORDING...";
    // ... recording implementation logic here
};

document.getElementById('b-file').onclick = () => document.getElementById('f-in').click();
document.getElementById('f-in').onchange = (e) => {
    const file = e.target.files[0];
    const reader = new FileReader();
    reader.onload = (re) => { 
        virtualCache = re.target.result; 
        diag.innerText = "📂 LOADED"; 
    };
    reader.readAsDataURL(file);
};

function stopAll() { 
    activeSources.forEach(s => { try{s.stop()}catch(e){} }); 
    activeSources = []; 
    diag.innerText = "STOPPED"; 
}

function slide(e, s) {
    if (isL) return;
    const r = e.currentTarget.getBoundingClientRect(); 
    const y = (e.touches ? e.touches[0].clientY : e.clientY);
    let v = 1 - (y - r.top) / r.height; 
    ints[s] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
}

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    const d = new Uint8Array(anal.frequencyBinCount); 
    anal.getByteTimeDomainData(d);
    ctx.clearRect(0, 0, cvs.width, cvs.height); 
    ctx.lineWidth = 4; ctx.strokeStyle = "#0f0"; 
    ctx.beginPath(); 
    const sw = cvs.width / d.length; 
    for (let i = 0; i < d.length; i++) {
        const y = (d[i] / 128.0) * cvs.height / 2;
        if (i === 0) ctx.moveTo(i * sw, y); else ctx.lineTo(i * sw, y); 
    }
    ctx.stroke();
}
