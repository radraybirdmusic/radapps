<script>
let aCtx, anal, src, rec, chks = [], recUrl = null, fileUrl = null, activeBucket = null;
let isL = false, isF = false, ints = {l:0.5, r:0.5}, isR = false, activeA = null, rawData = null;
const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d'), diag = document.getElementById('diag');
const bMic = document.getElementById('b-mic'), bRec = document.getElementById('b-rec'), bFile = document.getElementById('b-file'), fIn = document.getElementById('f-in'), bPlay = document.getElementById('b-play');

let playTimer;
bPlay.onpointerdown = () => { playTimer = setTimeout(triggerGlitch, 1000); };
bPlay.onpointerup = () => { clearTimeout(playTimer); };

function updatePlayUI(type) {
    activeBucket = type;
    bPlay.disabled = false;
    bPlay.style.opacity = 1;
    bPlay.innerText = "PLAY " + type + " ▶";
    bPlay.style.borderColor = (type === 'REC') ? 'var(--g)' : 'var(--p)';
}

bMic.onclick = async () => {
    document.getElementById('gate').style.display = 'none'; document.getElementById('ui').style.display = 'block';
    try {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        const s = await navigator.mediaDevices.getUserMedia({ audio: true });
        src = aCtx.createMediaStreamSource(s); anal = aCtx.createAnalyser(); anal.fftSize = 2048; src.connect(anal);
        rec = new MediaRecorder(s);
        rec.ondataavailable = e => { if (e.data.size > 0) chks.push(e.data); };
        rec.onstop = () => { 
            try {
                const blob = new Blob(chks, { type: 'audio/webm' });
                recUrl = URL.createObjectURL(blob); 
                diag.innerText = "REC SAVED: " + blob.size + " BYTES";
                updatePlayUI('REC');
            } catch (err) { diag.innerText = "REC ERROR: BLOB FAILED"; }
            isR = false; 
        };
        draw();
    } catch (err) { alert("Mic required."); }
};

bRec.onclick = () => {
    if (rec.state === "inactive") {
        const ring = document.getElementById('ring-path'), lab = document.getElementById('rec-l');
        let c = 4; const cd = setInterval(() => { lab.innerText = "WAIT " + c; if (c-- <= 0) {
            clearInterval(cd); lab.innerText = "RECORDING"; ring.style.display = "block"; chks = []; isR = true; rec.start(100); 
            let p = 0; const prg = setInterval(() => { p += 1.42; ring.style.strokeDashoffset = 251.3 - (251.3 * (p / 100));
            if (p >= 100) { clearInterval(prg); rec.stop(); ring.style.display = "none"; lab.innerText = "SET 🔒"; } }, 100);
        } }, 500);
    }
};

bFile.onclick = () => fIn.click();
fIn.onchange = async e => {
    const f = e.target.files[0]; if (!f) return;
    try {
        fileUrl = URL.createObjectURL(f);
        const ab = await f.arrayBuffer(); rawData = new DataView(ab);
        diag.innerText = "FILE LOADED: " + f.size + " BYTES";
        updatePlayUI('FILE');
        document.getElementById('file-l').innerText = "LOADED ✅";
    } catch (err) { diag.innerText = "LOAD ERROR: MEMORY LIMIT"; }
};

bPlay.onclick = () => {
    if (bPlay.innerText.includes("STOP")) { 
        if (activeA) { activeA.pause(); activeA.currentTime = 0; activeA.src = ""; } 
        bPlay.innerText = "PLAY " + (activeBucket || "") + " ▶"; 
        return; 
    }
    const target = (activeBucket === 'REC') ? recUrl : fileUrl;
    if (!target) { diag.innerText = "ERROR: BUCKET EMPTY"; return; }
    try {
        if (activeA) { activeA.pause(); activeA.src = ""; }
        activeA = new Audio(target); 
        activeA.play(); 
        bPlay.innerText = "STOP ■";
        activeA.onended = () => { bPlay.innerText = "PLAY " + activeBucket + " ▶"; };
    } catch (err) { diag.innerText = "PLAYBACK ERROR"; }
};

async function triggerGlitch() {
    let sourceData = rawData;
    if (!sourceData && recUrl) {
        try {
            const response = await fetch(recUrl);
            const ab = await response.arrayBuffer();
            sourceData = new DataView(ab);
        } catch(e) { diag.innerText = "GLITCH ERROR: SOURCE UNREADABLE"; }
    }
    if (!sourceData) { diag.innerText = "NO DATA TO RIP"; return; }
    
    try {
        diag.innerText = "💥 CHAOS RIP START";
        const len = 44100 * 2; const buf = aCtx.createBuffer(1, len, 44100); const data = buf.getChannelData(0);
        const offset = Math.floor(Math.random() * Math.max(0, sourceData.byteLength - len * 4));
        for (let i = 0; i < len; i++) {
            data[i] = (sourceData.getFloat32(offset + i * 4) % 2) - 1;
        }
        const s = aCtx.createBufferSource(); s.buffer = buf; s.connect(aCtx.destination); s.start();
        diag.innerText = "GLITCH ACTIVE // " + len + " SAMPLES";
    } catch (e) {
        diag.innerText = "CHAOS CRASH: CIRCUIT BROKEN";
    }
}

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    const d = new Uint8Array(anal.frequencyBinCount); anal.getByteTimeDomainData(d);
    ctx.clearRect(0, 0, cvs.width, cvs.height); ctx.lineWidth = 4; ctx.strokeStyle = isF ? "#0ff" : "#9d00ff"; ctx.beginPath();
    const sw = cvs.width / d.length; let x = 0;
    for (let i = 0; i < d.length; i++) {
        let amp = (isR || (activeA && !activeA.paused)) ? (d[i] / 128.0) : 1.0;
        const y = amp * cvs.height / 2; if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); x += sw;
    }
    ctx.stroke();
}

function slide(e, s) {
    if (isL || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect(); const y = e.touches ? e.touches[0].clientY : e.clientY;
    let v = 1 - (y - r.top) / r.height; ints[s] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
    if (ints.l > 1.0 || ints.r > 1.0) { document.body.classList.add('fae'); isF = true; } else { document.body.classList.remove('fae'); isF = false; }
}
function toggleLock() { isL = !isL; document.getElementById('p-lock').innerText = isL ? "LOCK: ON" : "LOCK: OFF"; }
