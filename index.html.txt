<script>
let aCtx, anal, src, rec, chks = [], virtualCache = null, isL = false, isF = false, ints = {l:0.5, r:0.5}, isR = false, activeSources = [];
let currentArpPitch = 1.0;
const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d'), diag = document.getElementById('diag'), prog = document.getElementById('rec-progress');
const bMic = document.getElementById('b-mic'), bRec = document.getElementById('b-rec'), bFile = document.getElementById('b-file'), fIn = document.getElementById('f-in'), bSave = document.getElementById('b-save');

bMic.onclick = async () => {
    document.getElementById('gate').style.display = 'none'; document.getElementById('ui').style.display = 'block';
    try {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        const s = await navigator.mediaDevices.getUserMedia({ audio: true });
        src = aCtx.createMediaStreamSource(s); anal = aCtx.createAnalyser(); anal.fftSize = 2048; 
        draw(); 
    } catch (err) { diag.innerText = "MIC ERROR"; }
};

bRec.onclick = () => {
    if (rec && rec.state === "recording") return;
    chks = []; isR = true;
    rec = new MediaRecorder(src.mediaStream);
    rec.ondataavailable = e => chks.push(e.data);
    rec.onstop = async () => {
        const blob = new Blob(chks, { type: 'audio/webm' });
        const reader = new FileReader();
        reader.onloadend = () => { virtualCache = reader.result; diag.innerText = `v2.05 CACHED`; };
        reader.readAsDataURL(blob);
        isR = false; currentArpPitch = 1.0;
    };
    rec.start();
    let startTime = Date.now();
    let interval = setInterval(() => {
        let elapsed = Date.now() - startTime;
        prog.style.width = Math.min((elapsed / 3500) * 100, 100) + '%';
        if (elapsed >= 3500) { clearInterval(interval); if(rec.state==="recording") rec.stop(); }
    }, 50);
};

bSave.onclick = () => {
    if(!virtualCache) { diag.innerText = "NOTHING TO SAVE"; return; }
    const link = document.createElement('a');
    link.href = virtualCache; link.download = `TRUMPET_GARDEN_${Date.now()}.webm`;
    link.click(); diag.innerText = "v2.05 SAVED";
};

bFile.onclick = () => fIn.click();
fIn.onchange = (e) => {
    const file = e.target.files[0];
    if(!file) return;
    const reader = new FileReader();
    reader.onload = (re) => { virtualCache = re.target.result; diag.innerText = "📂 LOADED"; };
    reader.readAsDataURL(file);
};

async function hamMomentum(multiplier) {
    if (!virtualCache) return;
    if (aCtx.state === 'suspended') await aCtx.resume();
    currentArpPitch *= multiplier;
    const res = await fetch(virtualCache);
    const ab = await res.arrayBuffer();
    aCtx.decodeAudioData(ab, (buffer) => {
        const s = aCtx.createBufferSource(); 
        s.buffer = buffer; 
        s.playbackRate.value = currentArpPitch;
        s.connect(anal); 
        anal.connect(aCtx.destination); 
        s.start(0);
        activeSources.push(s); 
    });
}

document.getElementById('p-rec-play').onclick = () => { currentArpPitch = 1.0; hamMomentum(1.0); };
document.getElementById('p-file-play').onclick = () => { currentArpPitch = 1.0; hamMomentum(1.0); };
document.getElementById('s-rec-stop').onclick = () => stopAll();
document.getElementById('s-file-stop').onclick = () => stopAll();

function stopAll() { activeSources.forEach(s => { try{s.stop()}catch(e){} }); activeSources = []; diag.innerText = "STOPPED"; }
function toggleLock() { isL = !isL; document.getElementById('p-lock').innerText = isL ? "LOCKED" : "LOCK"; }

function slide(e, s) {
    if (isL || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect(); const y = e.touches ? e.touches[0].clientY : e.clientY;
    let v = 1 - (y - r.top) / r.height; ints[s] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
    if (ints.l > 1.0 || ints.r > 1.0) { document.body.classList.add('fae'); isF = true; } else { document.body.classList.remove('fae'); isF = false; }
}

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    
    const timeData = new Uint8Array(anal.frequencyBinCount); 
    const freqData = new Uint8Array(anal.frequencyBinCount);
    anal.getByteTimeDomainData(timeData);
    anal.getByteFrequencyData(freqData);

    // Chakra Spectral Mapping (Freq to HSL)
    let avgFreq = freqData.reduce((a, b) => a + b) / freqData.length;
    let chakraHue = (avgFreq * 1.4) % 360; // Map freq to spectrum 0-360

    ctx.clearRect(0, 0, cvs.width, cvs.height); 
    ctx.lineWidth = 4; 
    
    // Line stays calm unless avgFreq > threshold
    ctx.strokeStyle = avgFreq > 5 ? `hsl(${chakraHue}, 100%, 50%)` : (isF ? "#000" : "#333"); 
    
    ctx.beginPath(); 
    const sw = cvs.width / timeData.length; 
    let x = 0;
    for (let i = 0; i < timeData.length; i++) {
        const v = timeData[i] / 128.0;
        const y = v * cvs.height / 2;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); 
        x += sw;
    }
    ctx.stroke();
}
