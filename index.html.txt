<script>
let aCtx, anal, masterBus, rec, chks = [], virtualCache = null, activeSources = [], isL = false, curPitch = 1.0, isF = false, ints = {l:0.5, r:0.5};

async function wake() {
    if (!aCtx) {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        anal = aCtx.createAnalyser(); anal.fftSize = 1024;
        masterBus = aCtx.createMediaStreamDestination();
        anal.connect(aCtx.destination);
        draw();
    }
    if (aCtx.state === 'suspended') await aCtx.resume();
}

async function pulse(e, pitch) {
    await wake();
    if (!virtualCache) return;
    const res = await fetch(virtualCache);
    const ab = await res.arrayBuffer();
    aCtx.decodeAudioData(ab, (buffer) => {
        const s = aCtx.createBufferSource();
        s.buffer = buffer; 
        s.playbackRate.value = pitch * curPitch;
        s.connect(anal); s.connect(masterBus);
        s.start(0); activeSources.push(s); 
    });
}

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d');
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    const t = new Uint8Array(anal.frequencyBinCount); anal.getByteTimeDomainData(t);
    ctx.clearRect(0, 0, cvs.width, cvs.height); ctx.lineWidth = 4;
    ctx.strokeStyle = isF ? "#000" : "#0f0";
    ctx.beginPath(); let sw = cvs.width / t.length;
    for (let i = 0; i < t.length; i++) {
        const y = (t[i] / 128.0) * cvs.height / 2;
        if (i === 0) ctx.moveTo(0, y); else ctx.lineTo(i * sw, y);
    }
    ctx.stroke();
}

function slide(e, s) {
    if (isL || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect();
    const y = e.touches ? e.touches[0].clientY : e.clientY;
    let v = 1 - (y - r.top) / r.height;
    ints[s] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
    if (ints.l > 1.0 || ints.r > 1.0) { document.body.classList.add('fae'); isF = true; } else { document.body.classList.remove('fae'); isF = false; }
}

function hamMomentum(m) { wake(); curPitch = m; document.getElementById('diag').innerText = `OCTAVE: ${m}`; }
function openFile() { document.getElementById('f-in').click(); }
document.getElementById('f-in').onchange = (e) => {
    const r = new FileReader(); r.onload = (re) => { virtualCache = re.target.result; document.getElementById('diag').innerText = "LOADED"; };
    r.readAsDataURL(e.target.files[0]);
};
function playCache() { pulse({target: document.querySelector('circle')}, 1.0); }
function stopAll() { activeSources.forEach(s => s.stop()); activeSources = []; }
function saveMaster() { if(virtualCache) { const l = document.createElement('a'); l.href = virtualCache; l.download = "master.webm"; l.click(); } }
function toggleLock() { isL = !isL; document.getElementById('p-lock').innerText = isL ? "LOCKED" : "LOCK"; }
</script>
