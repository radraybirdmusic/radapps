<script>
let aCtx, anal, masterBus, virtualCache = null, activeSources = [];

// Initialize Audio Context on Start
window.addEventListener('pointerdown', async () => {
    if (!aCtx) {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        anal = aCtx.createAnalyser(); 
        masterBus = aCtx.createMediaStreamDestination();
        anal.connect(aCtx.destination);
        draw();
    }
}, {once: true});

// 360 DEGREE JOYSTICK / VECTOR LOGIC
document.querySelectorAll('.v-pad').forEach(piston => {
    piston.addEventListener('pointermove', (e) => {
        if (e.buttons !== 1 && !e.touches) return;
        handleVector(e, piston);
    });
    piston.addEventListener('pointerdown', (e) => handleVector(e, piston));
    piston.addEventListener('pointerup', () => piston.setAttribute('stroke', '#fff'));
});

function handleVector(e, piston) {
    if (!virtualCache || !aCtx) return;
    const rect = piston.getBoundingClientRect();
    const cx = rect.left + rect.width/2, cy = rect.top + rect.height/2;
    const dx = e.clientX - cx, dy = e.clientY - cy;
    const dist = Math.sqrt(dx*dx + dy*dy);
    
    // Throttle / Vector calculation
    const throttle = Math.min(1.5, dist / (rect.width/2)); // 0 to 1.5 intensity
    const angle = Math.atan2(dy, dx); // 360 degree direction
    
    // Power the Signal
    strikeVector(parseFloat(piston.getAttribute('data-note')), throttle, angle);
    
    // Chakra Spectral Feedback
    const hue = (angle * (180/Math.PI) + 180);
    piston.setAttribute('stroke', `hsl(${hue}, 100%, 50%)`);
}

async function strikeVector(basePitch, throttle, angle) {
    if (aCtx.state === 'suspended') await aCtx.resume();
    const res = await fetch(virtualCache);
    const ab = await res.arrayBuffer();
    aCtx.decodeAudioData(ab, (buffer) => {
        const s = aCtx.createBufferSource();
        const g = aCtx.createGain();
        s.buffer = buffer;
        
        // Analog Frequency Shift based on Angle + Throttle
        s.playbackRate.value = basePitch + (Math.sin(angle) * (throttle * 0.2));
        
        // Voltage based on Throttle distance
        g.gain.setValueAtTime(throttle * 0.5, aCtx.currentTime);
        g.gain.exponentialRampToValueAtTime(0.001, aCtx.currentTime + 1.0);
        
        s.connect(g); g.connect(anal); g.connect(masterBus);
        s.start(0);
        activeSources.push(s);
    });
}

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    const cvs = document.getElementById('osc'); const ctx = cvs.getContext('2d');
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    const timeData = new Uint8Array(anal.frequencyBinCount);
    const freqData = new Uint8Array(anal.frequencyBinCount);
    anal.getByteTimeDomainData(timeData);
    anal.getByteFrequencyData(freqData);
    let avg = freqData.reduce((a, b) => a + b) / freqData.length;
    let hue = (avg * 2) % 360;
    ctx.clearRect(0, 0, cvs.width, cvs.height);
    ctx.lineWidth = 4;
    ctx.strokeStyle = avg > 3 ? `hsl(${hue}, 100%, 50%)` : "#333";
    ctx.beginPath(); let sw = cvs.width / timeData.length;
    for (let i = 0; i < timeData.length; i++) {
        const y = (timeData[i] / 128.0) * cvs.height / 2;
        if (i === 0) ctx.moveTo(0, y); else ctx.lineTo(i * sw, y);
    }
    ctx.stroke();
}

document.getElementById('b-file').onclick = () => document.getElementById('f-in').click();
document.getElementById('f-in').onchange = (e) => {
    const reader = new FileReader();
    reader.onload = (re) => { virtualCache = re.target.result; };
    reader.readAsDataURL(e.target.files[0]);
};
