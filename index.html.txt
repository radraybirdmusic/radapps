<script>
const ASSET_URL = "https://radraybirdmusic.github.io/radapps/buzz.wav";
let audioCtx, analyzer, source, recorder, chunks = [];
let audioBlob, audioUrl, isLock = false, isFae = false, intensities = {l:0.5, r:0.5}, isRecording = false;

const canvas = document.getElementById('oscillo');
const ctx = canvas.getContext('2d');
const btnMic = document.getElementById('btn-mic');
const btnRec = document.getElementById('btn-rec');
const btnFile = document.getElementById('btn-file');
const btnPlay = document.getElementById('btn-play');

btnMic.onclick = async () => {
    document.getElementById('gate').style.display = 'none';
    document.getElementById('ui').style.display = 'block';
    try {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        source = audioCtx.createMediaStreamSource(stream);
        analyzer = audioCtx.createAnalyser();
        analyzer.fftSize = 2048;
        source.connect(analyzer);
        recorder = new MediaRecorder(stream);
        recorder.ondataavailable = e => { if (e.data.size > 0) chunks.push(e.data); };
        recorder.onstop = () => {
            audioBlob = new Blob(chunks, { type: 'audio/wav' });
            audioUrl = URL.createObjectURL(audioBlob);
            btnPlay.disabled = false;
            btnPlay.style.opacity = 1;
            isRecording = false;
        };
        drawOscilloscope();
    } catch (err) { alert("Mic access denied."); }
};

btnRec.onclick = () => {
    if (recorder.state === "inactive") {
        const ring = document.getElementById('ring-path');
        const label = document.getElementById('rec-label');
        let count = 4;
        const countdown = setInterval(() => {
            label.innerText = "WAIT " + count;
            if (count-- <= 0) {
                clearInterval(countdown);
                label.innerText = "RECORDING";
                ring.style.display = "block";
                chunks = [];
                isRecording = true;
                recorder.start();
                let p = 0;
                const progress = setInterval(() => {
                    p += 1.42; 
                    ring.style.strokeDashoffset = 251.3 - (251.3 * (p / 100));
                    if (p >= 100) {
                        clearInterval(progress);
                        recorder.stop();
                        ring.style.display = "none";
                        label.innerText = "SET 🔒";
                    }
                }, 100);
            }
        }, 500);
    }
};

btnFile.onclick = async () => {
    if (audioCtx.state === 'suspended') await audioCtx.resume();
    const label = document.getElementById('file-label');
    label.innerText = "FETCHING...";
    try {
        const resp = await fetch(ASSET_URL, { cache: "no-store" });
        const buf = await resp.arrayBuffer();
        await audioCtx.decodeAudioData(buf.slice(0)); 
        audioBlob = new Blob([buf], { type: 'audio/wav' });
        audioUrl = URL.createObjectURL(audioBlob);
        btnPlay.disabled = false;
        btnPlay.style.opacity = 1;
        label.innerText = "LOADED ✅";
    } catch (e) { label.innerText = "FAIL ❌"; }
};

btnPlay.onclick = () => {
    if (!audioUrl) return;
    const audio = new Audio(audioUrl);
    audio.play();
    btnPlay.innerText = "STOP ■";
    audio.onended = () => { btnPlay.innerText = "PLAY ▶"; };
};

function drawOscilloscope() {
    requestAnimationFrame(drawOscilloscope);
    if (!analyzer) return;
    canvas.width = window.innerWidth; canvas.height = window.innerHeight;
    const bufferLength = analyzer.frequencyBinCount;
    const dataArray = new Uint8Array(bufferLength);
    analyzer.getByteTimeDomainData(dataArray);
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.lineWidth = 4;
    ctx.strokeStyle = isFae ? "#0ff" : "#9d00ff";
    ctx.beginPath();
    const sliceWidth = canvas.width / bufferLength;
    let x = 0;
    for (let i = 0; i < bufferLength; i++) {
        let amplitude = (isRecording || btnPlay.innerText.includes("STOP")) ? (dataArray[i] / 128.0) : 1.0;
        const y = amplitude * canvas.height / 2;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        x += sliceWidth;
    }
    ctx.stroke();
}

function slide(e, side) {
    if (isLock || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect();
    const y = e.touches ? e.touches[0].clientY : e.clientY;
    let v = 1 - (y - r.top) / r.height;
    intensities[side] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(side + '-fill').style.height = (intensities[side] / 1.5) * 100 + "%";
    if (intensities.l > 1.0 || intensities.r > 1.0) { document.body.classList.add('fae-mode'); isFae = true; }
    else { document.body.classList.remove('fae-mode'); isFae = false; }
}

function toggleLock() { 
    isLock = !isLock; 
    document.getElementById('p-lock').innerText = isLock ? "LOCK: ON" : "LOCK: OFF"; 
}
