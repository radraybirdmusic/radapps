<script>
    // CONFIG: Change this to your GitHub/Google Script URL
    const ASSET_URL = "https://raw.githubusercontent.com/your-repo/master/assets/buzz.wav";

    let aCtx, analyzer, isFae = false, isLock = false, intensities = {l:0.5, r:0.5};
    let audioBuffer, sourceNode;

    const cvs = document.getElementById('v-snake');
    const ctx = cvs.getContext('2d');

    function ignite() {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        document.getElementById('gate').style.display = 'none';
        document.getElementById('ui').style.display = 'block';
        draw();
    }

    async function fetchAndLock() {
        if (!aCtx) return;
        if (aCtx.state === 'suspended') await aCtx.resume();
        
        const status = document.getElementById('b-status');
        const icon = document.getElementById('b-icon');
        const ring = document.getElementById('ring-path');

        status.textContent = "FETCH";
        icon.textContent = "🌀";
        ring.style.display = "block";

        try {
            // Fetch external audio instead of recording locally
            const response = await fetch(ASSET_URL);
            const arrayBuffer = await response.arrayBuffer();
            audioBuffer = await aCtx.decodeAudioData(arrayBuffer);
            
            // Simulate the 7-second "Model" progress for UI consistency
            let p = 0;
            const rec = setInterval(() => {
                p += 5; // Faster visual fill for pre-loaded assets
                ring.style.strokeDashoffset = 282.7 - (282.7 * (p / 100));
                if (p >= 100) {
                    clearInterval(rec);
                    ring.style.display = "none";
                    icon.textContent = "🔒"; status.textContent = "SET";
                    document.getElementById('play-btn').style.opacity = 1;
                    document.getElementById('play-btn').style.pointerEvents = "auto";
                }
            }, 100);

        } catch (e) {
            alert("Asset Load Failed. Check CORS/URL.");
            status.textContent = "FAIL";
        }
    }

    function playback() {
        if (!audioBuffer) return;
        const btn = document.getElementById('play-btn');
        
        if (btn.innerText.includes("PLAY")) {
            sourceNode = aCtx.createBufferSource();
            sourceNode.buffer = audioBuffer;
            
            analyzer = aCtx.createAnalyser();
            sourceNode.connect(analyzer);
            analyzer.connect(aCtx.destination);
            
            sourceNode.start(0);
            btn.innerText = "STOP ■";
            sourceNode.onended = () => { btn.innerText = "PLAY ▶"; };
        } else {
            if (sourceNode) sourceNode.stop();
            btn.innerText = "PLAY ▶";
        }
    }

    function slide(e, side) {
        if (isLock || (e.buttons !== 1 && !e.touches)) return;
        const r = e.currentTarget.getBoundingClientRect();
        const y = e.touches ? e.touches[0].clientY : e.clientY;
        let v = 1 - (y - r.top) / r.height;
        intensities[side] = Math.max(0, Math.min(1.5, v * 1.5));
        document.getElementById(side + '-fill').style.height = (intensities[side] / 1.5) * 100 + "%";
        if (Math.max(intensities.l, intensities.r) > 1.0) { document.body.classList.add('fae-mode'); isFae = true; }
        else { document.body.classList.remove('fae-mode'); isFae = false; }
    }

    function toggleLock() { isLock = !isLock; document.getElementById('p-lock').innerText = isLock ? "LOCK: ON" : "LOCK: OFF"; }

    function draw() {
        cvs.width = window.innerWidth; cvs.height = window.innerHeight;
        ctx.clearRect(0,0,cvs.width, cvs.height);
        if (analyzer) {
            const data = new Uint8Array(analyzer.frequencyBinCount);
            analyzer.getByteTimeDomainData(data);
            ctx.strokeStyle = isFae ? '#0ff' : '#9d00ff';
            ctx.lineWidth = 4; ctx.beginPath();
            for(let i=0; i<data.length; i++) {
                let x = (i/data.length) * cvs.width;
                let y = (data[i]/128.0) * (cvs.height/8) + (cvs.height/2);
                if (i===0) ctx.moveTo(x,y); else ctx.lineTo(x,y);
            }
            ctx.stroke();
        }
        requestAnimationFrame(draw);
    }
</script>
