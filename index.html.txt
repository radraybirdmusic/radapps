<script>
let aCtx, anal, src, rec, chks = [], recUrl = null, fileUrl = null, isL = false, isF = false, ints = {l:0.5, r:0.5}, isR = false, activeA = null, activeNode = null, rawData = null;
const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d'), diag = document.getElementById('diag');
const bMic = document.getElementById('b-mic'), bRec = document.getElementById('b-rec'), bFile = document.getElementById('b-file'), fIn = document.getElementById('f-in'), bGlitch = document.getElementById('b-glitch');

bMic.onclick = async () => {
    document.getElementById('gate').style.display = 'none'; document.getElementById('ui').style.display = 'block';
    try {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        const s = await navigator.mediaDevices.getUserMedia({ audio: true });
        src = aCtx.createMediaStreamSource(s); anal = aCtx.createAnalyser(); anal.fftSize = 2048; 
        draw();
    } catch (err) { alert("Mic required."); }
};

bRec.onclick = () => {
    if (rec && rec.state === "recording") return;
    chks = []; isR = true; src.connect(anal);
    rec = new MediaRecorder(src.mediaStream);
    rec.ondataavailable = e => chks.push(e.data);
    rec.onstop = () => {
        const b = new Blob(chks, { type: 'audio/webm' });
        recUrl = URL.createObjectURL(b);
        diag.innerText = "🐝 RECORDED: " + b.size + " BYTES";
        src.disconnect(anal); isR = false;
    };
    rec.start();
    setTimeout(() => { if(rec.state==="recording") rec.stop(); }, 3000);
};

bFile.onclick = () => fIn.click();
fIn.onchange = async e => {
    const f = e.target.files[0]; if (!f) return;
    fileUrl = URL.createObjectURL(f);
    const ab = await f.arrayBuffer(); rawData = new DataView(ab);
    diag.innerText = "📂 LOADED: " + f.size + " BYTES";
};

document.getElementById('p-rec-play').onclick = () => play(recUrl);
document.getElementById('p-rec-stop').onclick = () => stop();
document.getElementById('p-file-play').onclick = () => play(fileUrl);
document.getElementById('p-file-stop').onclick = () => stop();

function play(url) {
    if (!url) return; stop();
    activeA = new Audio(url);
    activeNode = aCtx.createMediaElementSource(activeA);
    activeNode.connect(anal); anal.connect(aCtx.destination);
    activeA.play();
    activeA.onended = () => stop();
}

function stop() {
    if (activeA) { activeA.pause(); activeA.currentTime = 0; }
    if (activeNode) { activeNode.disconnect(); activeNode = null; }
    diag.innerText = "AUDIO TERMINATED";
}

bGlitch.onclick = async () => {
    let d = rawData;
    if (!d && recUrl) {
        const res = await fetch(recUrl); const ab = await res.arrayBuffer(); d = new DataView(ab);
    }
    if (!d) { diag.innerText = "GLITCH ERROR: NO DATA"; return; }
    diag.innerText = "💥 GLITCHING RAW DATA...";
    const len = 44100 * 1.5; const buf = aCtx.createBuffer(1, len, 44100); const dat = buf.getChannelData(0);
    const off = Math.floor(Math.random() * (d.byteLength - len * 4));
    for (let i = 0; i < len; i++) { try { dat[i] = (d.getFloat32(off + i * 4) % 2) - 1; } catch(e) { dat[i] = 0; } }
    const s = aCtx.createBufferSource(); s.buffer = buf; s.connect(aCtx.destination); s.start();
};

function draw() {
    requestAnimationFrame(draw); if (!anal) return;
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    const d = new Uint8Array(anal.frequencyBinCount); anal.getByteTimeDomainData(d);
    ctx.clearRect(0, 0, cvs.width, cvs.height); ctx.lineWidth = 4; ctx.strokeStyle = isF ? "#0ff" : "#9d00ff";
    if (!(isR || (activeA && !activeA.paused))) return;
    ctx.beginPath(); const sw = cvs.width / d.length; let x = 0;
    for (let i = 0; i < d.length; i++) {
        const y = (d[i] / 128.0) * cvs.height / 2;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); x += sw;
    }
    ctx.stroke();
}

function slide(e, s) {
    if (isL || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect(); const y = e.touches ? e.touches[0].clientY : e.clientY;
    let v = 1 - (y - r.top) / r.height; ints[s] = Math.max(0, Math.min(1.5, v * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
    if (ints.l > 1.0 || ints.r > 1.0) { document.body.classList.add('fae'); isF = true; } else { document.body.classList.remove('fae'); isF = false; }
}
function toggleLock() { isL = !isL; document.getElementById('p-lock').innerText = isL ? "LOCK: ON" : "LOCK: OFF"; }
