<script>
const bridge = new BroadcastChannel('tinyguy_bridge');
let aCtx, anal, src, rec, chks = [], virtualCache = null, isL = false, isF = false, ints = {l:0.5, r:0.5}, activeSources = [];
let currentArpPitch = 1.0;
let frazzle = 1.0; // ROAD RATTLE SYNC

window.onload = async () => {
    try {
        aCtx = new (window.AudioContext || window.webkitAudioContext)();
        const s = await navigator.mediaDevices.getUserMedia({ audio: true });
        src = aCtx.createMediaStreamSource(s); anal = aCtx.createAnalyser(); 
    } catch (e) { diag.innerText = "NEXUS OFFLINE"; }
};

// TINYGUY COMMAND: MOMENTUM & FRAZZLE
async function hamMomentum(multiplier) {
    if (!virtualCache) return;
    currentArpPitch *= multiplier;
    frazzle = 2.5 * currentArpPitch; // Tension Multiplier logic
    
    const res = await fetch(virtualCache);
    const ab = await res.arrayBuffer();
    aCtx.decodeAudioData(ab, (buffer) => {
        const s = aCtx.createBufferSource();
        s.buffer = buffer; s.playbackRate.value = currentArpPitch;
        s.connect(anal); anal.connect(aCtx.destination);
        s.start(0);
        activeSources.push(s);
        bridge.postMessage({ header: { protocol: "AURORA_NEXUS" }, payload: { action: "HAMMER_SLAM", params: { frequency: currentArpPitch * 528 } } });
    });
}

function draw() {
    requestAnimationFrame(draw);
    if (!anal) return;
    const cvs = document.getElementById('osc'), ctx = cvs.getContext('2d');
    cvs.width = window.innerWidth; cvs.height = window.innerHeight;
    
    // THE FRAZZLE SHAKE (Road Sync)
    let shakeX = (Math.random() - 0.5) * frazzle;
    let shakeY = (Math.random() - 0.5) * frazzle;
    
    const d = new Uint8Array(anal.frequencyBinCount); anal.getByteTimeDomainData(d);
    ctx.clearRect(0, 0, cvs.width, cvs.height);
    ctx.save();
    ctx.translate(shakeX, shakeY);
    ctx.lineWidth = 4; ctx.strokeStyle = isF ? "#000" : "#00FFFF";
    ctx.shadowBlur = 15; ctx.shadowColor = "#00FFFF";
    
    ctx.beginPath();
    let x = 0; const sw = cvs.width / d.length;
    for (let i = 0; i < d.length; i++) {
        let y = (d[i] / 128.0) * cvs.height / 2;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        x += sw;
    }
    ctx.stroke();
    ctx.restore();
    if(frazzle > 1.0) frazzle *= 0.98; // Natural Decay
}
function slide(e, s) {
    if (isL || (e.buttons !== 1 && !e.touches)) return;
    const r = e.currentTarget.getBoundingClientRect(); const y = e.touches ? e.touches[0].clientY : e.clientY;
    ints[s] = Math.max(0, Math.min(1.5, (1 - (y - r.top) / r.height) * 1.5));
    document.getElementById(s + '-f').style.height = (ints[s] / 1.5) * 100 + "%";
    isF = (ints.l > 1.0 || ints.r > 1.0); document.body.classList.toggle('fae', isF);
}
function toggleLock() { isL = !isL; document.getElementById('p-lock').innerText = isL ? "LOCK: ON" : "LOCK: OFF"; }
draw();
