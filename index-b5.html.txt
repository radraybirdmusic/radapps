<script>
const bridge = new BroadcastChannel('tinyguy_bridge');
let aCtx, isTrumpet = false, frazzle = 1.0, currentPitch = 1.0;
const oscs = new Map(), pointers = new Map();
const canvas = document.getElementById('mainCanvas'), ctx = canvas.getContext('2d');

// THE 18-BUCKET PHASE MAP
const buckets = [164.81, 196.00, 220.00, 246.94, 261.63, 293.66, 329.63, 349.23, 392.00];

function toggleTrumpet() {
    isTrumpet = !isTrumpet;
    document.getElementById('instBtn').innerText = isTrumpet ? "🎺TRP_MODE" : "🎹KBD_STATE";
    killAll();
}

function hamSlam(mult) {
    currentPitch *= mult;
    frazzle = 8.0; // TRIGGER ROAD RATTLE
    bridge.postMessage({
        header: { protocol: "AURORA_NEXUS", source: "FORGE_MASTER" },
        payload: { action: "HAMMER_SLAM", params: { frequency: 528 * currentPitch, intensity: frazzle } }
    });
}

function slide(e, s) {
    if (e.buttons !== 1 && e.type !== 'pointerdown') return;
    const r = e.currentTarget.getBoundingClientRect();
    const val = Math.max(0, Math.min(1, (1 - (e.clientY - r.top) / r.height)));
    document.getElementById(s + '-f').style.height = (val * 100) + "%";
    if (val > 0.9) document.body.classList.add('fae');
    else if (val < 0.1) document.body.classList.remove('fae');
}

function handleDown(e) {
    if (e.target.closest('#ui') || e.target.closest('.sld')) return;
    if (!aCtx) aCtx = new (window.AudioContext || window.webkitAudioContext)();
    
    let cX = window.innerWidth / 2, cY = window.innerHeight / 2;
    let idx = Math.floor(Math.random() * 9); // Placeholder for grid logic
    
    const g = aCtx.createGain();
    const o = aCtx.createOscillator();
    o.type = isTrumpet ? 'sawtooth' : 'triangle';
    o.frequency.setValueAtTime(buckets[idx] * currentPitch, aCtx.currentTime);
    g.gain.setTargetAtTime(0.2, aCtx.currentTime, 0.05);
    
    o.connect(g); o.start(); g.connect(aCtx.destination);
    oscs.set(e.pointerId, {o, g, x: e.clientX, y: e.clientY});
}

function handleUp(e) {
    if (oscs.has(e.pointerId)) {
        let v = oscs.get(e.pointerId);
        v.g.gain.setTargetAtTime(0, aCtx.currentTime, 0.05);
        setTimeout(() => { v.o.stop(); oscs.delete(e.pointerId); }, 100);
    }
}

function killAll() { oscs.forEach(v => v.o.stop()); oscs.clear(); }

function animate() {
    canvas.width = window.innerWidth; canvas.height = window.innerHeight;
    ctx.clearRect(0,0,canvas.width,canvas.height);
    
    // THE FRAZZLE SHAKE
    let sX = (Math.random() - 0.5) * frazzle;
    let sY = (Math.random() - 0.5) * frazzle;
    ctx.save();
    ctx.translate(sX, sY);

    // DRAW KINEMATIC GRID
    ctx.strokeStyle = "#00FFFF";
    ctx.shadowBlur = 15;
    ctx.shadowColor = "#00FFFF";
    
    oscs.forEach((v, id) => {
        // Tension Strings
        ctx.beginPath();
        ctx.setLineDash([5, 5]);
        ctx.moveTo(canvas.width/2, canvas.height);
        ctx.lineTo(v.x, v.y);
        ctx.stroke();
        ctx.setLineDash([]);
        
        // Active Node
        ctx.beginPath();
        ctx.arc(v.x, v.y, 40, 0, Math.PI*2);
        ctx.stroke();
    });

    ctx.restore();
    if (frazzle > 1.0) frazzle *= 0.95; // Rattle decay
    requestAnimationFrame(animate);
}

canvas.addEventListener('pointerdown', handleDown);
canvas.addEventListener('pointerup', handleUp);
animate();
</script>
